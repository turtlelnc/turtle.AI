cmake_minimum_required(VERSION 3.25)
project(turtle_mlx LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TURTLE_BUILD_MLX "Build the MLX model executable" ON)
option(TURTLE_FETCH_MLX "Fetch pinned MLX 0.31.2 instead of an installed package" OFF)
option(TURTLE_CAPACITY_MOE "Use capacity-limited MoE and load-balancing loss" ON)
option(TURTLE_MLX_SMOKE_TESTS "Run real CPU training/ELF smoke tests" ON)

if(WIN32 AND TURTLE_BUILD_MLX)
  message(FATAL_ERROR "This MLX application supports macOS/Linux. Use the root CPU trainer on Windows.")
endif()

find_package(Threads REQUIRED)
add_library(turtle_mlx_bpe BPE.cpp)
target_include_directories(turtle_mlx_bpe PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

if(TURTLE_BUILD_MLX)
  if(TURTLE_FETCH_MLX)
    include(FetchContent)
    set(MLX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(MLX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(MLX_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(mlx
      URL https://github.com/ml-explore/mlx/archive/68cf2fddd8de5edd8ab3d926391772b2e2cedad8.tar.gz
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(mlx)
  else()
    find_package(MLX 0.31.2 CONFIG REQUIRED)
  endif()
  add_executable(turtle-mlx main.cpp)
  target_link_libraries(turtle-mlx PRIVATE mlx turtle_mlx_bpe Threads::Threads)
  target_include_directories(turtle-mlx PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
  if(TURTLE_CAPACITY_MOE)
    target_compile_definitions(turtle-mlx PRIVATE USE_CAPACITY_MOE)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mlx-bpe-tests tests/bpe_tests.cpp)
  target_link_libraries(mlx-bpe-tests PRIVATE turtle_mlx_bpe Threads::Threads)
  add_test(NAME mlx-bpe COMMAND mlx-bpe-tests)
  if(TURTLE_BUILD_MLX AND TURTLE_MLX_SMOKE_TESTS)
    add_executable(mlx-model-checks tests/model_checks.cpp)
    target_link_libraries(mlx-model-checks PRIVATE mlx turtle_mlx_bpe Threads::Threads)
    target_include_directories(mlx-model-checks PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
    if(TURTLE_CAPACITY_MOE)
      target_compile_definitions(mlx-model-checks PRIVATE USE_CAPACITY_MOE)
    endif()
    add_test(NAME mlx-model-numerics COMMAND mlx-model-checks)
    add_test(NAME mlx-model-numerics-fast COMMAND mlx-model-checks
      --fast-sdpa --gather-sparse --split-qkv --generic-rms)
    set_tests_properties(mlx-model-numerics PROPERTIES TIMEOUT 120)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME mlx-runtime COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE:turtle-mlx>)
    set_tests_properties(mlx-runtime PROPERTIES TIMEOUT 600)
  endif()
endif()
